cmake_minimum_required(VERSION 3.13)
project(native_surfaces LANGUAGES CXX)
set(PLUGIN_NAME "native_surfaces_plugin")
find_package(PkgConfig REQUIRED)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
add_library(${PLUGIN_NAME} SHARED "native_surfaces_plugin.cc")
apply_standard_settings(${PLUGIN_NAME})
set_target_properties(${PLUGIN_NAME} PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
target_compile_features(${PLUGIN_NAME} PUBLIC cxx_std_17)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter PkgConfig::EPOXY ${CMAKE_DL_LIBS})
set(native_surfaces_bundled_libraries "" PARENT_SCOPE)
