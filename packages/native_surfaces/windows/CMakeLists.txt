cmake_minimum_required(VERSION 3.14)
project(native_surfaces LANGUAGES CXX)
set(PLUGIN_NAME "native_surfaces_plugin")
add_library(${PLUGIN_NAME} SHARED "native_surfaces_plugin.cpp")
apply_standard_settings(${PLUGIN_NAME})
target_compile_features(${PLUGIN_NAME} PUBLIC cxx_std_17)
target_compile_definitions(${PLUGIN_NAME} PRIVATE NATIVE_SURFACES_PLUGIN_IMPL)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter flutter_wrapper_plugin)
set(native_surfaces_bundled_libraries "" PARENT_SCOPE)
