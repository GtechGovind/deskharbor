cmake_minimum_required(VERSION 3.18)
project(deskharbor_texture_boundary_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
option(DH_NATIVE_SANITIZE "Enable native boundary sanitizers" OFF)
foreach(name copy lease presentation library_path owned_snapshot renderer_health)
  add_executable(${name}_test ${name}_test.cpp)
  target_compile_features(${name}_test PRIVATE cxx_std_17)
  target_link_libraries(${name}_test PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${name}_test PRIVATE /W4 /WX)
    if(DH_NATIVE_SANITIZE)
      target_compile_options(${name}_test PRIVATE /fsanitize=address /Zi)
    endif()
  else()
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Werror)
    if(DH_NATIVE_SANITIZE)
      target_compile_options(${name}_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${name}_test PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME ${name}_test COMMAND ${name}_test)
endforeach()

# The Android JNI translation unit includes this same header with these flags.
# Compile-only: a dummy runtime test would not exercise loader behavior.
add_library(loader_header_compile OBJECT loader_header_compile.cpp)
target_compile_features(loader_header_compile PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(loader_header_compile PRIVATE /W4 /WX)
else()
  target_compile_options(loader_header_compile PRIVATE -Wall -Wextra -Werror)
endif()
